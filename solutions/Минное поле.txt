#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    string fields[n];
    for (int i = 0; i < n; ++i)
        cin >> fields[i];
    int ans = 0;
    for (int i = 0; i < n; ++i)
        for (auto let : fields[i])
            if (let == '*')
                ++ans;
    cout << ans << '\n';
    return 0;
}
