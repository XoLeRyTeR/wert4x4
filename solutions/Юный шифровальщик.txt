#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector <int> a(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    for (int i = 0; i < m; ++i) {
        for (auto& x : a)
            if (x) --x;
        a.push_back(a.size() % 2 == 0 ? a.back() : a.front());
    }
    for (auto x : a)
        cout << x << ' ';
    cout << '\n';
    return 0;
}
