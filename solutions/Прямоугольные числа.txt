#include <bits/stdc++.h>

using namespace std;

string solve(int n) {
    vector <int> nums;
    while (n != 0) {
        nums.push_back(n % 10);
        n /= 10;
    }

    if (nums[0] != nums[2] || nums[1] != nums[3])
        return "no";

    if (nums[0] == nums[3])
        return "square " + to_string(nums[0]) + ' ' + to_string(nums[0]);

    return "rectangle " + to_string(nums[0]) + ' ' + to_string(nums[1]);
}

int main() {
    int a1, a2;
    cin >> a1 >> a2;
    cout << solve(a1) << '\n';
    cout << solve(a2) << '\n';
    return 0;
}
