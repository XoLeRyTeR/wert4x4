#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    bool a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    int ans = 0, b = false;
    for (int i = 1; i < n; ++i) {
        if (!a[i - 1] && a[i])
            b = true;
        if (b && a[i - 1] && !a[i]) {
            ++ans;
            b = false;
        }
    }
    cout << ans << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
