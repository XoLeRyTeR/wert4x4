#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int f[n];
    for (int i = 0; i < n; ++i)
        cin >> f[i];
    int T;
    cin >> T;
    int ind = 0;
    while (ind < n && f[ind] <= T) {
        T -= f[ind];
        ++ind;
    }
    cout << ind << '\n';
    return 0;
}
