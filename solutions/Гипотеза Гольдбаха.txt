#include <bits/stdc++.h>

using namespace std;

bool is_prime(int n) {
    const int m = sqrt(n);
    for (int i = 2; i <= m; ++i)
        if (n % i == 0)
            return false;
    return true;
}

int main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n / 2; ++i)
        if (is_prime(i) && is_prime(n - i))
            cout << i << ' ' << n - i << '\n';
    return 0;
}
