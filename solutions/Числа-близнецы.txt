#include <bits/stdc++.h>

using namespace std;

bool is_prime(int n) {
    const int l = sqrt(n);
    for (int i = 2; i <= l; ++i)
        if (n % i == 0)
            return false;
    return true;
}

int main() {
    int a, b;
    cin >> a >> b;
    for (int i = max(2, a); i <= b - 2; ++i)
        if (is_prime(i) && is_prime(i + 2))
            cout << i << ' ' << i + 2 << '\n';
    return 0;
}
