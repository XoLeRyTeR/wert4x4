#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    int ans = 1, len = 1;
    for (int i = 1; i < n; ++i)
        if (a[i - 1] == a[i])
            ++len;
        else {
            ans = max(ans, len);
            len = 1;
        }
    cout << max(ans, len) << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
