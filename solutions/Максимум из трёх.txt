#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    for (int i = 0; i < n; i += 3)
        cout << max({a[i], a[i + 1], a[i + 2]}) << ' ';
    cout << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
