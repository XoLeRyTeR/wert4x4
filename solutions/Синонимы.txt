#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    map <string, string> v;
    for (int i = 0; i < n; ++i) {
        string a, b;
        cin >> a >> b;
        v[a] = b;
        v[b] = a;
    }
    for (int i = 0; i < q; ++i) {
        string s;
        cin >> s;
        cout << (v.find(s) != v.end() ? v[s] : s) << '\n';
    }
    return 0;
}
