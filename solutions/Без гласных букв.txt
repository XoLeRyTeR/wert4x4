#include <bits/stdc++.h>

using namespace std;

set <char> vowels = { 'a', 'e', 'i', 'o', 'u', 'y' };

string clear_vowel(const string& w) {
    string res;
    for (auto let : w)
        if (!vowels.count(let))
            res += let;
    return res;
}

int main() {
    int n, m;
    cin >> n >> m;
    map <string, string> v;
    for (int i = 0; i < n; ++i) {
        string a;
        cin >> a;
        v[clear_vowel(a)] = a;
    }
    for (int i = 0; i < m; ++i) {
        string a;
        cin >> a;
        cout << v[a] << ' ';
    }
    cout << '\n';
    return 0;
}
