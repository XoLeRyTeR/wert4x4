#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    map <int, int> ans;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        ++ans[a[i]];
    }
    cout << ans.begin()->second << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
