#include <bits/stdc++.h>

using namespace std;

int main() {
    int a, b;
    cin >> a >> b;
    vector <int> ans;
    for (int i = 10; i <= 99; ++i)
        if (i % a == 0 && i % b != 0)
            ans.push_back(i);
    if (ans.empty())
        cout << "NO\n";
    else {
        for (auto x : ans)
            cout << x << ' ';
        cout << '\n';
    }
    return 0;
}
