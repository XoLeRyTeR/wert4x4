#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n], ans = 0;
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    double m = accumulate(a, a + n, 0.0) / n;
    for (int i = 0; i < n; ++i)
        if (a[i] < m)
            ++ans;
    cout << ans << '\n';
    return 0;
}
