#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int t[4] = {};
    for (int i = 0; i < n; ++i) {
        int x, y;
        cin >> x >> y;
        if (x < 0) {
            if (y < 0)
                ++t[2];
            else
                ++t[0];
        }
        else {
            if (y < 0)
                ++t[3];
            else
                ++t[1];
        }
    }
    if (t[0] > 2 || t[1] > 9 || t[2] > 5 || t[3] > 9 || t[0] * 10 + t[1] > 23 || t[2] * 10 + t[3] > 59) {
        cout << "Broken\n";
        return 0;
    }
    cout << t[0] << t[1] << ' ' << t[2] << t[3] << '\n';
    return 0;
}
