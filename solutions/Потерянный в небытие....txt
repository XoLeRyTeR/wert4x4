#include <bits/stdc++.h>

using namespace std;

int main() {
    string s[5];
    for (int i = 0; i < 5; ++i)
        cin >> s[i];
    for (int i = 0; i < 5; ++i)
        if (s[i].size() == 11 && s[i].substr(0, 4) == "8912" && s[i].substr(9, 2) == "39") {
            cout << s[i] << '\n';
            return 0;
        }
    return 0;
}
