#include <bits/stdc++.h>

using namespace std;

bool is_prime(int n) {
    for (int i = 2; i <= sqrt(n); ++i)
        if (n % i == 0)
            return false;
    return true;
}

int main() {
    int n;
    cin >> n;
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        int a;
        cin >> a;
        if (is_prime(a))
            ++ans;
    }
    cout << ans << '\n';
    return 0;
}
