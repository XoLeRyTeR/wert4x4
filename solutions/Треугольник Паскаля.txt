#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    cout << "1\n";
    int m = 3;
    vector <int> a = {0, 1, 0}, b;
    for (int i = 1; i < n; ++i) {
        b = vector<int>(1, 0);
        for (int i = 1; i < m; ++i)
            b.push_back(a[i - 1] + a[i]);
        b.push_back(0);

        for (int i = 1; i < m; ++i)
            cout << b[i] << ' ';
        cout << '\n';

        ++m;
        a = b;
    }
    return 0;
}
