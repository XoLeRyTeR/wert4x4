#include <bits/stdc++.h>
#define all(X) X.begin(), X.end()

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    set <int> values(a, a + n);
    vector <int> ans(all(values));
    cout << ans[ans.size() - 2] << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
