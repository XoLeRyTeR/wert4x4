#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int v[n], t[n];
    for (int i = 0; i < n; ++i)
        cin >> v[i];
    for (int i = 0; i < n; ++i)
        cin >> t[i];
    int volume = accumulate(v, v + n, 0);
    double temperature = 0;
    for (int i = 0; i < n; ++i)
        temperature += t[i] * v[i];
    temperature /= volume;
    cout << volume << ' ' << temperature << '\n';
    return 0;
}
