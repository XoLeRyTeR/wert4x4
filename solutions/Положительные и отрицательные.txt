#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n], pos = 0, neg = 0;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
        if (a[i] > 0)
            pos += a[i];
        else
            neg += a[i];
    }
    cout << pos << '\n' << neg << '\n';
    cout << a[0];
    for (int i = 1; i < n; ++i)
        cout << ' ' << a[i];
    cout << '\n';
    return 0;
}
