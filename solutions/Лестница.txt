#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    int w = 0, lim = 1;
    for (int i = 0; i < n - 1; ++i) {
        cout << a[i];
        if (++w == lim) {
            cout << '\n';
            w = 0;
            ++lim;
        }
        else
            cout << ' ';
    }
    cout << a[n - 1] << '\n';
    cout << a[0];
    for (int i = 1; i < n; ++i)
        cout << ' ' << a[i];
    cout << '\n';
    return 0;
}
