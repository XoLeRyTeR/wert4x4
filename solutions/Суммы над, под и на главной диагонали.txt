#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n][n];
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j)
            cin >> a[i][j];
    int sums[3] = {0, 0, 0};
    for (int i = 0; i < n; ++i)
        for (int j = i + 1; j < n; ++j) {
            sums[0] += a[i][j];
            sums[2] += a[j][i];
        }
    for (int i = 0; i < n; ++i)
        sums[1] += a[i][i];
    for (int i = 0; i < 3; ++i)
        cout << sums[i] << '\n';
    return 0;
}
