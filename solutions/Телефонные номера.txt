#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    map <int, string> ans;
    for (int i = 0; i < n; ++i) {
        string name; int num;
        cin >> name >> num;
        ans[num] = name;
    }
    int num;
    cin >> num;
    if (ans.count(num) != 0)
        cout << ans[num] << '\n';
    else
        cout << num << '\n';
    return 0;
}
