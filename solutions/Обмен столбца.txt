#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    int a[n][m];
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            cin >> a[i][j];
    int x = 0, v = a[0][0];
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            if (a[i][j] > v) {
                v = a[i][j];
                x = j;
            }
    for (int i = 0; i < n; ++i)
        swap(a[i][0], a[i][x]);
    for (int i = 0; i < n; ++i, cout << '\n')
        for (int j = 0; j < m; ++j)
            cout << a[i][j] << ' ';
    return 0;
}
