#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    int x = 0, y = 0, z = 0;
    for (int i = 0; i < n; ++i) {
        if (a[i] == 0)
            ++z;
        if (a[i] > 0)
            ++x;
        if (a[i] < 0)
            ++y;
    }
    cout << x << '\n' << y << '\n' << z << '\n';
    for (int i = 0; i < n; ++i)
        cout << a[i] << ' ';
    cout << '\n';
    return 0;
}
