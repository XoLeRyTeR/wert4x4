#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    map <string, int> words;
    for (int i = 0; i < n; ++i) {
        string word;
        cin >> word;
        ++words[word];
    }
    vector <pair <int, string>> w;
    for (const auto& [key, value] : words)
        w.emplace_back(value, key);
    sort(w.begin(), w.end());
    cout << w.back().second << '\n';
    return 0;
}
