#include <bits/stdc++.h>

using namespace std;

bool is_prime(int n) {
    const int l = sqrt(n);
    for (int i = 2; i <= l; ++i)
        if (n % i == 0)
            return false;
    return true;
}

int main() {
    int n;
    cin >> n;
    int x = 1;
    while (n != 0)
        if (is_prime(++x))
            --n;
    cout << x << '\n';
    return 0;
}
