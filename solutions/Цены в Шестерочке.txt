#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    map <string, int> ans;
    for (int i = 0; i < n; ++i) {
        string name; int price;
        cin >> name >> price;
        ans[name] = min(ans[name] == 0 ? price : ans[name], price);
    }
    cout << ans.size() << '\n';
    for (const auto& [name, price] : ans)
        cout << name << ' ' << price << '\n';
    return 0;
}
