#include <bits/stdc++.h>

using namespace std;

int main() {
    long long n, q;
    cin >> n >> q;
    map <int, int> a;
    for (int i = 0; i < q; ++i) {
        string c; long long index;
        cin >> c >> index;
        if (c == "SET") {
            long long value;
            cin >> value;
            a[index] = value;
        }
        else
            cout << a[index] << '\n';
    }
    return 0;
}
