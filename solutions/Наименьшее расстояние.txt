#include <bits/stdc++.h>

using namespace std;

struct Point {
    int x, y;
};

double area(Point A, Point B, Point C) {
    return abs((B.x - A.x) * (C.y - A.y) - (C.x - A.x) * (B.y - A.y)) / 2.0;
}

double dist(Point A, Point B) {
    auto dx = A.x - B.x;
    auto dy = A.y - B.y;
    return sqrt(dx * dx + dy * dy);
}

int main() {
    Point A, B, C, D;
    cin >> A.x >> A.y >> B.x >> B.y >> C.x >> C.y >> D.x >> D.y;

    auto h1 = area(A, B, D) / dist(A, B) * 2;
    auto h2 = area(B, C, D) / dist(B, C) * 2;
    auto h3 = area(C, A, D) / dist(A, C) * 2;

    cout << fixed << setprecision(6) << min({h1, h2, h3}) << '\n';
    return 0;
}
