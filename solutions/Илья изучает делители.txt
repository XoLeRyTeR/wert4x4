#include <bits/stdc++.h>

using namespace std;

int divcount(int n) {
    int res = 0;
    for (int i = 1; i <= n; ++i)
        if (n % i == 0)
            ++res;
    return res;
}

int main() {
    int n;
    cin >> n;
    for (int i = 2; i <= n; ++i)
        cout << i << " has " << divcount(i) << " divisors\n";
    return 0;
}
