#include <bits/stdc++.h>

using namespace std;

int main() {
    string n;
    cin >> n;
    int sum = 0;
    const int len = n.size();
    for (int i = 0; i < len; ++i)
        sum += n[i] - '0';
    int right = 0;
    for (int i = len - 1; i > -1; --i) {
        right += n[i] - '0';
        if (2 * right == sum) {
            cout << i << '\n';
            return 0;
        }
    }
    return 0;
}
