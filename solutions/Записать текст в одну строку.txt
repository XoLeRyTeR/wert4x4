#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n; cin.ignore();
    string line;
    vector <string> ans;
    for (int i = 0; i < n; ++i) {
        getline(cin, line);
        stringstream ss(line);
        string word;
        while (ss >> word)
            ans.push_back(word);
    }
    cout << ans.size() << '\n';
    for (const auto& word : ans)
        cout << word << ' ';
    cout << '\n';
    return 0;
}
