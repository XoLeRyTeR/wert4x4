#include <bits/stdc++.h>

using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    string names[n];
    for (int i = 0; i < n; ++i)
        cin >> names[i];
    sort(names, names + n);
    vector <string> g[k];
    for (int i = 0; i < n; ++i)
        g[i % k].push_back(names[i]);
    for (int i = 0; i < k; ++i) {
        cout << g[i].size() << '\n';
        for (const auto& name : g[i])
            cout << name << '\n';
    }
    return 0;
}
