#include <bits/stdc++.h>

using namespace std;

int solve(int r) {
    switch (r % 3) {
        case 0:
            return r;
        case 1:
            return r - 1;
        case 2:
            return r + 1;
    }
}

int main() {
    int r;
    cin >> r;
    cout << solve(r) << '\n';
    return 0;
}
