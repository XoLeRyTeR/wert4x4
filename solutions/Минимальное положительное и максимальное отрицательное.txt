#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    int a1 = 1e5, a2 = -1e5;
    for (int i = 0; i < n; ++i) {
        if (a[i] == 0)
            continue;
        if (a[i] > 0)
            a1 = min(a1, a[i]);
        else
            a2 = max(a2, a[i]);
    }
    cout << a1 << ' ' << a2 << '\n';
    return 0;
}
