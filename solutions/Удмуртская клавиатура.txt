#include <bits/stdc++.h>
#define all(X) X.begin(), X.end()

using namespace std;

int main() {
    string x;
    cin >> x; cin.ignore();
    set <char> lets(all(x));
    lets.insert(' ');
    string line;
    getline(cin, line);
    int ans = 0;
    for (auto let : line)
        ans += (lets.count(let) == 0 ? 6 : 1);
    cout << ans << '\n';
    return 0;
}
