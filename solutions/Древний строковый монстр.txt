#include <bits/stdc++.h>

using namespace std;

string f(string s) {
    int n = s.size();
    string lets[2];
    for (int i = 0; i < n; ++i)
        lets[i % 2] += s[i];
    s = lets[1] + lets[0];
    s[1] = 'x';
    s[3] = 'y';
    s[5] = 'z';
    s.erase(6, 2);
    s.insert(6, s.substr(6, 3));
    s.erase(0, 1);
    return s;
}

int main() {
    int n, k;
    cin >> n >> k;
    string s(n, ' ');
    for (int i = 0; i < n; ++i)
        s[i] = 'a' + i;
    for (int i = 0; i < k; ++i)
        s = f(s);
    cout << s << '\n';
    return 0;
}
