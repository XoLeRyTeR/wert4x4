#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

ll G(ll x) {
    return (4 * x + 42) % 137;
}

ll P(ll x) {
    return x % 2 ? x * G(707) + G(2 * x) : 2 * G(x + 5);
}

ll F(ll x) {
    return P(x) * P(x + 1) + 45 * P(3 * x) * G(x + 8);
}

int main() {
    ll x;
    cin >> x;
    cout << F(x) << '\n';
    return 0;
}
