#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    int a[n], b[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i];
    for (int i = 0; i < n; ++i)
        cin >> b[i];
    int ans = a[0] / b[0];
    for (int i = 1; i < n; ++i)
        if (b[i] != 0)
            ans = min(ans, a[i] / b[i]);
    cout << ans << '\n';
    return 0;
}
