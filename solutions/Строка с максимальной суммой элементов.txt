using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    int a[n][m];
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < m; ++j)
            cin >> a[i][j];
    int ans = 0, sum = accumulate(a[0], a[0] + m, 0);
    for (int i = 1; i < n; ++i)
        if (int s = accumulate(a[i], a[i] + m, 0); s > sum) {
            ans = i;
            sum = s;
        }
    cout << ans + 1 << '\n';
    return 0;
}
