#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    int ans = s[0] - '0';
    const int n = s.size();
    for (int i = 1; i < n; i += 2)
        switch (s[i]) {
        case '-':
            ans -= s[i + 1] - '0';
            break;
        case '+':
            ans += s[i + 1] - '0';
            break;
        default:
            break;
        }
    cout << ans << '\n';
    return 0;
}
