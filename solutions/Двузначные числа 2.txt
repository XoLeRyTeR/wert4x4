#include <bits/stdc++.h>

using namespace std;

int numsum(int n) {
    int res = 0;
    while (n) {
        res += n % 10;
        n /= 10;
    }
    return res;
}

int main() {
    int n;
    cin >> n;
    for (int i = 10; i <= 99; ++i)
        if (numsum(i) == n)
            cout << i << ' ';
    cout << '\n';
    return 0;
}
