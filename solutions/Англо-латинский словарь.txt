#include <bits/stdc++.h>

using namespace std;

int main() {
    int n;
    cin >> n;
    char _;
    map <string, set <string>> a;
    for (int i = 0; i < n; ++i) {
        string s, w;
        cin >> s >> _;
        while (true) {
            cin >> w;
            if (w.back() == ',')
                w.pop_back();
            else {
                a[w].insert(s);
                break;
            }
            a[w].insert(s);
        }
    }
    cout << a.size() << '\n';
    for (const auto& [key, value] : a) {
        cout << key << " - ";
        const auto last = --value.end();
        for (auto it = value.begin(); it != last; ++it)
            cout << *it << ", ";
        cout << *last << '\n';
    }
    return 0;
}
