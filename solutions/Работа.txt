#include <bits/stdc++.h>

using namespace std;

int main() {
    int n = 3;
    vector <int> ans;
    for (int i = 0; i < n; ++i) {
        int t, s;
        cin >> t >> s;
        if (t <= 8 || s / t >= 5000)
            ans.push_back(i + 1);
    }
    cout << ans.size() << '\n';
    for (auto x : ans)
        cout << x  << ' ';
    cout << '\n';
    return 0;
}
