#include <bits/stdc++.h>

using namespace std;

int main() {
    string s;
    cin >> s;
    int b = 0;
    for (auto let : s)
        if (let == '(')
            ++b;
        else
            if (--b < 0) {
                cout << "INCORRECT\n";
                return 0;
            }
    cout << (b == 0 ? "CORRECT\n" : "INCORRECT\n");
    return 0;
}
