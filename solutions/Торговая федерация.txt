#include <bits/stdc++.h>

using namespace std;
typedef long long ll;

int main() {
    int n;
    cin >> n;
    ll a[n], b[n], c[n];
    for (int i = 0; i < n; ++i)
        cin >> a[i] >> b[i] >> c[i];
    ll sum = 0;
    for (int i = 0; i < n; ++i)
        sum += a[i] * b[i] * c[i];
    sort(a, a + n);
    sort(b, b + n);
    sort(c, c + n);
    ll opt_sum = 0;
    for (int i = 0; i < n; ++i)
        opt_sum += a[i] * b[i] * c[i];
    cout << sum << ' ' << opt_sum << '\n';
    return 0;
}
